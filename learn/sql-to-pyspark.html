If you know SQL, you already know most of PySpark

PySpark's DataFrame API isn't a different language to learn from scratch — it's the same relational concepts (select, filter, group, join, window) wrapped in Python method calls instead of keywords. Once you can see the 1:1 mapping, "translating" a query becomes mechanical rather than mysterious.

Why the practice site runs pandas, not real PySpark: real PySpark needs a Spark cluster (a JVM), which can't run inside a browser tab. So the PySpark practice questions execute equivalent pandas code live, and show the real PySpark syntax alongside it for reference. This page uses real PySpark syntax throughout, since that's what you'll actually write on the job or in an interview.
Selecting & Filtering
Select specific columns
SQL
SELECT name, salary
FROM employees;
PySpark
df.select("name", "salary")
Filter rows (WHERE)
SQL
SELECT * FROM employees
WHERE salary > 70000;
PySpark
df.filter(F.col("salary") > 70000)
# .where(...) is just an alias for .filter(...)
Remove duplicates
SQL
SELECT DISTINCT department
FROM employees;
PySpark
df.select("department").distinct()
Sorting & Limiting
Sort results (ORDER BY)
SQL
SELECT * FROM employees
ORDER BY salary DESC;
PySpark
df.orderBy(F.col("salary").desc())
Limit row count
SQL
SELECT * FROM employees
LIMIT 5;
PySpark
df.limit(5)
Columns & Renaming
Alias / rename a column
SQL
SELECT salary AS annual_salary
FROM employees;
PySpark
df.withColumnRenamed("salary", "annual_salary")
# or: df.select(F.col("salary").alias("annual_salary"))
Add a calculated column
SQL
SELECT *, salary * 12 AS monthly
FROM employees;
PySpark
df.withColumn("monthly", F.col("salary") * 12)
Grouping & Aggregating
GROUP BY + aggregate
SQL
SELECT department, COUNT(*)
FROM employees
GROUP BY department;
PySpark
df.groupBy("department").count()
Filter groups (HAVING)
SQL
SELECT department, COUNT(*) c
FROM employees
GROUP BY department
HAVING COUNT(*) > 2;
PySpark
df.groupBy("department").count() \\
  .filter(F.col("count") > 2)
# HAVING is just another .filter(), applied after groupBy
Joins
Inner join
SQL
SELECT * FROM orders o
JOIN customers c
  ON o.customer_id = c.customer_id;
PySpark
orders.join(customers, on="customer_id")
Left join
SQL
SELECT * FROM customers c
LEFT JOIN orders o
  ON c.customer_id = o.customer_id;
PySpark
customers.join(orders, on="customer_id", how="left")
Conditional Logic & NULLs
CASE WHEN
SQL
SELECT name,
  CASE WHEN salary >= 80000
    THEN 'Senior' ELSE 'Junior' END
FROM employees;
PySpark
df.withColumn("level",
  F.when(F.col("salary") >= 80000, "Senior")
   .otherwise("Junior"))
COALESCE (default value)
SQL
SELECT COALESCE(manager_id, 0)
FROM employees;
PySpark
F.coalesce(F.col("manager_id"), F.lit(0))
NULL checks
SQL
SELECT * FROM employees
WHERE manager_id IS NULL;
PySpark
df.filter(F.col("manager_id").isNull())
Window Functions
RANK within a group
SQL
SELECT name,
  RANK() OVER (
    PARTITION BY department
    ORDER BY salary DESC)
FROM employees;
PySpark
w = Window.partitionBy("department") \\
           .orderBy(F.col("salary").desc())
df.withColumn("rnk", F.rank().over(w))
Running total
SQL
SELECT *,
  SUM(amount) OVER (ORDER BY order_date)
FROM orders;
PySpark
w = Window.orderBy("order_date") \\
           .rowsBetween(Window.unboundedPreceding, Window.currentRow)
df.withColumn("running_total", F.sum("amount").over(w))
Combining Queries
UNION ALL
SQL
SELECT * FROM a
UNION ALL
SELECT * FROM b;
PySpark
a.unionByName(b)
CTE (WITH clause)
SQL
WITH dept_avg AS (
  SELECT department, AVG(salary) a
  FROM employees GROUP BY department
)
SELECT * FROM dept_avg;
PySpark
dept_avg = df.groupBy("department") \\
              .agg(F.avg("salary").alias("a"))
# a Python variable IS your CTE — keep chaining off it.
# or: df.createOrReplaceTempView("employees"); spark.sql("...")
